// UI preview screenshots for design review (goldens).
// Run: flutter test test/ui_preview_test.dart --update-goldens
// Output: test/goldens/ui_*.png  (phone: 390x844 @3x)
import 'dart:io';

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:cricket_scorer_app/main.dart';

// Golden screenshot generation needs >10 min (full app compile + boot).

/// Load a real TTF as Roboto so golden screenshots render readable text.
Future<void> _loadTestFonts() async {
  Future<void> load(String path, FontWeight w) async {
    final data = await File(path).readAsBytes();
    final loader = FontLoader('Roboto')
      ..addFont(Future.value(ByteData.view(data.buffer)));
    await loader.load();
  }

  await load('/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf', FontWeight.normal);
  await load('/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf', FontWeight.bold);
}

/// Pump and swallow known RenderFlex overflow errors (real UI bug, tracked
/// separately) so golden screenshots can still be captured.
Future<void> _settle(WidgetTester tester) async {
  await tester.pumpAndSettle();
  dynamic e = tester.takeException();
  while (e != null) {
    e = tester.takeException();
  }
}

Future<void> _bootApp(WidgetTester tester) async {
  await _loadTestFonts();
  tester.view.physicalSize = const Size(1170, 2532);
  tester.view.devicePixelRatio = 3.0;
  SharedPreferences.setMockInitialValues({});
  await tester.pumpWidget(const Score60App());
  await tester.pump();
  await tester.pump(const Duration(milliseconds: 1600));
  await _settle(tester);
  // Dismiss the first-run welcome dialog.
  final later = find.text('Add Past Scores Later');
  if (later.evaluate().isNotEmpty) {
    await tester.tap(later);
    await _settle(tester);
  }
}

void main() {
  testWidgets('UI preview: main tabs', (tester) async {
    addTearDown(tester.view.resetPhysicalSize);
    addTearDown(tester.view.resetDevicePixelRatio);
    await _bootApp(tester);

    // 1. Home
    await expectLater(
      find.byType(HomeShell),
      matchesGoldenFile('goldens/ui_home.png'),
    );

    // 2. Matches tab
    await tester.tap(find.text('Matches'));
    await _settle(tester);
    await expectLater(
      find.byType(HomeShell),
      matchesGoldenFile('goldens/ui_matches.png'),
    );

    // 3. Statistics tab
    await tester.tap(find.text('Statistics'));
    await _settle(tester);
    await expectLater(
      find.byType(HomeShell),
      matchesGoldenFile('goldens/ui_stats.png'),
    );

    // 4. More drawer
    await tester.tap(find.text('More'));
    await _settle(tester);
    await expectLater(
      find.byType(HomeShell),
      matchesGoldenFile('goldens/ui_drawer.png'),
    );
  });

  testWidgets('UI preview: pushed screens', (tester) async {
    addTearDown(tester.view.resetPhysicalSize);
    addTearDown(tester.view.resetDevicePixelRatio);
    await _bootApp(tester);

    // 5. Teams screen (via Home action card)
    await tester.tap(find.text('Teams'));
    await _settle(tester);
    await expectLater(
      find.byType(Scaffold),
      matchesGoldenFile('goldens/ui_teams.png'),
    );
  });
}
